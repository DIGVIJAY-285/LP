import numpy as np
import matplotlib.pyplot as plt


def f(x):
    return (x + 3) ** 2

def grad(x):
    return 2 * (x + 3)      # derivative of (x+3)^2



x = 2               # starting point
lr = 0.1            # learning rate (step size)
tol = 1e-6          # stop when the step becomes this small
max_iter = 1000     # safety limit so the loop can never run forever
path = [x]          # store every x so we can plot the journey later




for i in range(max_iter):
    step = lr * grad(x)
    x = x - step
    path.append(x)
    if abs(step) < tol:
        break


print(f'Iterations      : {i + 1}')
print(f'Local minimum at: x = {x:.4f}')
print(f'Minimum value   : y = {f(x):.6f}')



xs = np.linspace(-6, 4, 200)
p = np.array(path)

plt.plot(xs, f(xs), label='y = (x+3)²')
plt.scatter(p, f(p), c='red', s=20, label='gradient descent steps')
plt.xlabel('x'); plt.ylabel('y'); plt.legend(); plt.grid(True)
plt.show()




def run(lr, start=2, n=50):
    x = start
    for _ in range(n):
        x = x - lr * grad(x)
    return x

for lr in [0.01, 0.1, 0.9, 1.1]:
    print(f'lr = {lr:<5} -> x after 50 steps = {run(lr):.4f}')
