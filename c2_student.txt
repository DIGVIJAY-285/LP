pragma solidity ^0.8.34;

contract StudentData {

    // Structure to store student details
    struct Student {
        uint id;
        string name;
        uint age;
        string department;
    }

    // Array to store students
    Student[] public students;

    function addStudent(
        uint _id,
        string memory _name,
        uint _age,
        string memory _department
    ) public {
        students.push(Student(_id, _name, _age, _department));
    }

    function getStudent(uint _index)
        public
        view
        returns (uint, string memory, uint, string memory)
    {
        Student memory s = students[_index];

        return (s.id, s.name, s.age, s.department);
    }

    function updateStudent(
        uint _index,
        uint _id,
        string memory _name,
        uint _age,
        string memory _department
    ) public {
        require(_index < students.length, "Invalid index");

        students[_index].id = _id;
        students[_index].name = _name;
        students[_index].age = _age;
        students[_index].department = _department;
    }

    function getTotalStudents() public view returns (uint) {
        return students.length;
    }
}
